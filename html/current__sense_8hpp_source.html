620 PhaseValues<float> get_phase_currents(bool filter = true)
621 {
622 if (!aligned_) {
623 nu_log::push(nu_log::Level::WARN, nu_log::Id::CS_NOT_ALIGNED);
624 return {};
625 }
626
627 std::array<float, N> amps{};
628
629 if (dma_mode_) {
630 amps = read_dma_sensors();
631 } else if (filter) {
632 amps = read_filtered_sensors();
633 } else {
634 amps = read_sensors();
635 }
636
637 PhaseValues<float> phase_amps{0.f, 0.f, 0.f};
638
639 if (phase_idx_.a > -1) {
640 phase_amps.a = phase_dirs_.a * amps.at(phase_idx_.a);
641 }
642 if (phase_idx_.b > -1) {
643 phase_amps.b = phase_dirs_.b * amps.at(phase_idx_.b);
644 }
645 if (phase_idx_.c > -1) {
646 phase_amps.c = phase_dirs_.c * amps.at(phase_idx_.c);
647 }
648
649 if constexpr (N == 3) {
650 // KCL correction: subtract the shared offset error from each phase.
651 // For an ideal star-connected load, I_a + I_b + I_c = 0. Any non-zero
652 // sum is common-mode sensor bias; distributing it equally across all
653 // three phases is the minimum-variance correction that enforces the constraint.
654 const float kcl_error = (phase_amps.a + phase_amps.b + phase_amps.c) / 3.f;
655 phase_amps.a -= kcl_error;
656 phase_amps.b -= kcl_error;
657 phase_amps.c -= kcl_error;
658 return phase_amps;
659 }
660
661 // 2-sensor mode: KCL reconstruction of the unmeasured phase.
662 if (phase_idx_.a == -1) {
663 phase_amps.a = -phase_amps.b - phase_amps.c;
664 }
665 if (phase_idx_.b == -1) {
666 phase_amps.b = -phase_amps.a - phase_amps.c;
667 }
668 if (phase_idx_.c == -1) {
669 phase_amps.c = -phase_amps.a - phase_amps.b;
670 }
671
672 return phase_amps;
673 }

749 bool init_dma_sensors(const pwm_hal::FlexPWMPin & trigger_source)
750 {
751 bool all_ok = true;
752 bool adc1_hw_trigger_configured = false;
753
754 for (uint8_t i = 0; i < static_cast<uint8_t>(N); ++i) {
755 const int arduino_pin = sensors_.at(i)->get_pin();
756
757 // Resolve pin to ADC channel
758 const uint8_t ch1 = adc_dma_hal::arduino_pin_to_adc1_channel(
759 static_cast<uint8_t>(arduino_pin));
760 const uint8_t ch2 = adc_dma_hal::arduino_pin_to_adc2_channel(
761 static_cast<uint8_t>(arduino_pin));
762
763 const bool use_adc2 = (ch1 == 255u && ch2 != 255u);
764 const uint8_t adc_ch = use_adc2 ? ch2 : ch1;
765
766 if (adc_ch == 255u) {
767 nu_log::push(nu_log::Level::WARN, nu_log::Id::CS_DMA_PIN_INVALID,
768 static_cast<float>(arduino_pin));
769 nu_log::drain(Serial);
770 all_ok = false;
771 continue;
772 }
773
774 if (!use_adc2) {
775 // Configure ADC1 hardware-trigger mode once (shared by all ADC1 sensors)
776 if (!adc1_hw_trigger_configured) {
777 adc_dma_hal::configure_adc1_hardware_trigger(adc_ch);
778 adc1_hw_trigger_configured = true;
779 }
780 // Route FlexPWM trigger → XBAR → ADC1_ETC_TRIG[i]
781 adc_dma_hal::connect_flexpwm_trigger_to_adc1_trig(trigger_source, i);
782 // Configure ADC_ETC TRIG[i] for this channel
783 adc_dma_hal::configure_adc_etc_trig(i, adc_ch);
784 } else {
785 // ADC2 sensor (e.g., A9): use existing ADC2 path with TRIG1 in ADC_ETC
786 adc_dma_hal::configure_adc2_hardware_trigger(adc_ch);
787 adc_dma_hal::connect_flexpwm_trigger_to_adc2(trigger_source);
788 adc_dma_hal::configure_adc_etc_trigger2(adc_ch);
789 }
790
791 // Arm the DMA channel for this sensor
792 all_ok &= sensors_.at(i)->init_dma(adc_ch, i, use_adc2);
793 }
794
795 dma_mode_ = all_ok;
796 return all_ok;
797 }
