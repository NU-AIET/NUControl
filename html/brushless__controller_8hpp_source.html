315 RLIdentResult identify_rl(bool update_params = true,
316 float test_voltage = 1.f,
317 int n_L_pulses = 200,
318 float L_pulse_us = 200.f,
319 int r_avg_samples = 128)
320 {
321 // ------------------------------------------------------------------
322 // Setup
323 // ------------------------------------------------------------------
324 const float V_bias = 1.f; // Keeps PWM count > 0 on all pins at all times
325
326 driver_.enable();
327
328 // Positive configuration: drive phase A high, B and C at bias
329 auto V_pos = PhaseValues<float>{V_bias + test_voltage, V_bias, V_bias};
330 // Negative configuration: drive phase A at bias, B and C high
331 auto V_neg = PhaseValues<float>{V_bias, V_bias + test_voltage, V_bias + test_voltage};
332 // Idle: all phases at bias (zero differential)
333 auto V_idle = PhaseValues<float>{V_bias, V_bias, V_bias};
334
335 // ------------------------------------------------------------------
336 // Step 1: Resistance — DC steady-state measurement
337 // ------------------------------------------------------------------
338 driver_.set_phase_voltages(V_pos);
339 delay(20); // wait >5τ; safe upper bound for any motor with τ < 4 ms
340
341 float I_sum[3] = {0.f, 0.f, 0.f};
342 for (int s = 0; s < r_avg_samples; ++s) {
343 auto raw = cs_.read_unaligned();
344 I_sum[0] += raw[0];
345 I_sum[1] += raw[1];
346 I_sum[2] += raw[2];
347 delayMicroseconds(50);
348 }
349
350 driver_.set_phase_voltages(V_idle);
351
352 // Average and find which sensor carries the most current (phase A sensor)
353 float I_avg[3];
354 float I_abs_max = 0.f;
355 int driven_sensor = 0;
356 for (int i = 0; i < 3; ++i) {
357 I_avg[i] = I_sum[i] / static_cast<float>(r_avg_samples);
358 if (fabsf(I_avg[i]) > I_abs_max) {
359 I_abs_max = fabsf(I_avg[i]);
360 driven_sensor = i;
361 }
362 }
363
364 // For Y-connected motor with V_A = V_test, V_B = V_C = 0:
365 // I_A = (2/3) * V_test / R_phase → R_phase = (2/3) * V_test / I_A
366 float R_phase = (2.f / 3.f) * test_voltage / I_abs_max;
367
368 // Let the motor demagnetise before the L test
369 delay(5);
370
371 // ------------------------------------------------------------------
372 // Step 2: Inductance — ODrive-style alternating voltage pulses
373 // ------------------------------------------------------------------
374 // Skip the first few pulses while the current oscillation stabilises
375 const int skip_pulses = 5;
376
377 float dI_sum = 0.f;
378 int valid = 0;
379
380 for (int p = 0; p < n_L_pulses + skip_pulses; ++p) {
381 // Positive half-pulse: I_A rises
382 driver_.set_phase_voltages(V_pos);
383 delayMicroseconds(static_cast<uint32_t>(L_pulse_us));
384 float I_peak = cs_.read_unaligned()[driven_sensor];
385
386 // Negative half-pulse: I_A falls
387 driver_.set_phase_voltages(V_neg);
388 delayMicroseconds(static_cast<uint32_t>(L_pulse_us));
389 float I_valley = cs_.read_unaligned()[driven_sensor];
390
391 if (p >= skip_pulses) {
392 dI_sum += (I_peak - I_valley);
393 ++valid;
394 }
395 }
396
397 driver_.set_phase_voltages(V_idle);
398 driver_.disable();
399
400 // ------------------------------------------------------------------
401 // Compute L_phase
402 // ------------------------------------------------------------------
403 // In steady-state: ΔI = I_peak − I_valley = 2 * V_test * t_pulse / L_eff
404 // L_eff = 3/2 * L_phase (Y-connected, phase A vs B||C)
405 // → L_phase = (2/3) * L_eff = (2/3) * (2 * V_test * t_pulse) / ΔI
406 float dI_avg = (valid > 0) ? (dI_sum / static_cast<float>(valid)) : 0.f;
407 float t_pulse = L_pulse_us * 1e-6f;
408 float L_phase = (valid > 0) ? ((2.f / 3.f) * 2.f * test_voltage * t_pulse / dI_avg) : 0.f;
409
410 // ------------------------------------------------------------------
411 // Sanity checks
412 // ------------------------------------------------------------------
413 bool ok = (I_abs_max > 0.02f) // sensor responding
414 && (R_phase > 0.f)
415 && (dI_avg > 0.005f) // L measurement above noise floor
416 && (L_phase > 0.f);
417
418 RLIdentResult result{R_phase, L_phase, ok};
419
420 nu_log::push(nu_log::Level::INFO, nu_log::Id::RL_RESULT_R, R_phase);
421 nu_log::push(nu_log::Level::INFO, nu_log::Id::RL_RESULT_L, L_phase * 1e6f);
422 nu_log::push(nu_log::Level::INFO, nu_log::Id::RL_RESULT_DI, dI_avg);
423 nu_log::push(nu_log::Level::INFO, nu_log::Id::RL_RESULT_OK, ok ? 1.f : 0.f);
424 nu_log::drain(Serial);
425
426 // ------------------------------------------------------------------
427 // Apply results to controller state (if requested and valid)
428 // ------------------------------------------------------------------
429 if (update_params && ok) {
430 motor_.phase_R = R_phase;
431 motor_.phase_L = L_phase;
432
433 // Recompute default PI gains from new RL values
434 Kp_ = motor_.phase_L * _2_PI_ * 25.f;
435 Ki_ = motor_.phase_R * _2_PI_ * 25.f;
436 set_feedback_control(
437 PIController<QuadDirectValues<float>>(Kp_, Ki_, control_period_s_));
438
439 // Rebuild feedforward filter with new R and L
440 ff_filter_ = MotorFeedforward(motor_.phase_R, motor_.phase_L, control_period_s_);
441 }
442
443 return result;
444 }

473 bool align_sensors()
474 {
475 e_ang_offset_ = 0.f;
476
477 auto ret = cs_.align_sensors(driver_, 0.5f * motor_.phase_R * motor_.SAFE_CURRENT);
478 if(!ret){
479 nu_log::push(nu_log::Level::ERROR, nu_log::Id::ALIGN_FAIL);
480 nu_log::drain(Serial);
481 return ret;
482 }

485 update_sensors();
486 open_loop_shaft_angle_ = 0.f;
487 open_loop_shaft_velocity_ = 0.f;
488
489 float init_ang = encoder_angle.get_full_angle();
490
491 float offset_sum = 0.f;
492 float offset_pos_sum = 0.f;
493 float offset_neg_sum = 0.f;
494 int samples = 0;
495
496 nu_log::push(nu_log::Level::INFO, nu_log::Id::ALIGN_SCAN_START, 1.f);
497 nu_log::drain(Serial);
498
499 // float inital_shaft_angle = shaft_angle_.get_full_angle();
500 set_control_mode(ControllerMode::OPEN_LOOP_VELOCITY);
501 target_ = static_cast<float>(calibration_dir_) * calibration_scan_speed_;
502 start_control(1000, false);
503 while(static_cast<float>(calibration_dir_) * open_loop_shaft_angle_ < calibration_scan_distance_){
504 control_step();
505 float elec_cmd = get_eangle(open_loop_shaft_angle_);
506 float elec_meas = get_eangle(static_cast<float>(calibration_dir_) * pos_sensor_dir_ * encoder_angle.get_full_angle());
507 float offset_pos = normalize_angle(elec_cmd - elec_meas);
508 float offset_neg = normalize_angle(elec_cmd + elec_meas);
509 offset_pos_sum += offset_pos;
510 offset_neg_sum += offset_neg;
511 samples++;
512 delay(1);
513 }
514 stop_control();
515
516 float new_ang = encoder_angle.get_full_angle();
517
518 if(new_ang > init_ang + 0.1){
519 pos_sensor_dir_ = calibration_dir_;
520 offset_sum = offset_pos_sum;
521
522
523 } else {
524 if(new_ang < init_ang - 0.1) {
525 pos_sensor_dir_ = -calibration_dir_;
526 offset_sum = offset_neg_sum;
527 }
528 else{
529 nu_log::push(nu_log::Level::ERROR, nu_log::Id::NO_MOTION_DETECTED, init_ang);
530 nu_log::push(nu_log::Level::INFO, nu_log::Id::ALIGN_FINAL_ANGLE, new_ang);
531 nu_log::drain(Serial);
532 return false;
533 }
534 }
535 nu_log::push(nu_log::Level::INFO, nu_log::Id::ENCODER_DIRECTION,
536 static_cast<float>(pos_sensor_dir_));
537 nu_log::push(nu_log::Level::INFO, nu_log::Id::ALIGN_SCAN_START, -1.f);
538 nu_log::drain(Serial);
539
540 open_loop_shaft_angle_ = 0.f;
541 target_ = static_cast<float>(calibration_dir_) * -calibration_scan_speed_;
542 start_control(1000, false);
543 while(static_cast<float>(calibration_dir_) * open_loop_shaft_angle_ > -calibration_scan_distance_){
544 control_step();
545 float elec_cmd = get_eangle(open_loop_shaft_angle_);
546 float elec_meas = get_eangle(static_cast<float>(calibration_dir_) * pos_sensor_dir_ * encoder_angle.get_full_angle());
547 float offset = normalize_angle(elec_cmd - elec_meas);
548 offset_sum += offset;
549 samples++;
550 delay(1);
551 }
552 stop_control();
553 shaft_velocity_ = 0;
554 update_sensors();
555 stop_control();
556 e_ang_offset_ = normalize_angle(offset_sum / samples);
557
558 nu_log::push(nu_log::Level::INFO, nu_log::Id::ZERO_ELEC_ANGLE, e_ang_offset_);
559 nu_log::drain(Serial);
560
561 return true;

621 void start_control(int control_period_us, bool use_internal_timer = true)
622 {
623 // kT = 0 would produce infinite current requests in update_control().
624 // Catch the misconfiguration at startup rather than silently at runtime.
625 if (motor_.kT <= 0.f) {
626 nu_log::push(nu_log::Level::ERROR, nu_log::Id::FAULT_DRIVER_INIT, motor_.kT);
627 nu_log::drain(Serial);
628 return;
629 }
630
631 control_period_us_ = control_period_us;
632 control_period_s_ = control_period_us_ * 1e-6;
633 control_freq_hz_ = 1.f / control_period_s_;
634
635 set_filters(filter_cutoff_freq_hz_, filter_cutoff_freq_hz_current_, filter_cutoff_freq_hz_fb_);
636
637 // Reconstruct feedforward filter at the new control period
638 ff_filter_ = MotorFeedforward(motor_.phase_R, motor_.phase_L, control_period_s_);
639
640 // Read the encoder via the legacy filter path regardless of PLL mode so that
641 // shaft_angle_ is valid before any PLL or filter state is initialised.
642 // Using the PLL here would produce a large transient because pos_est_ is still
643 // at its default (0 rad) and the error would be the full motor position.
644 {
645 bool pll_was_active = use_velocity_pll_;
646 use_velocity_pll_ = false;
647 update_sensors();
648 use_velocity_pll_ = pll_was_active;
649 }
650
651 if (use_velocity_pll_) {
652 velocity_pll_.reset(shaft_angle_, 0.f);
653 } else {
654 pos_filter_.reset(shaft_angle_);
655 vel_filter_.reset(shaft_angle_);
656 vel_filter_cutoff_.reset();
657 }
658
659 shaft_velocity_ = 0.f;
660
661 last_error_ = QuadDirectValues<float>{0.f, 0.f};
662 last_command_ = QuadDirectValues<float>{0.f, 0.f};
663
664 driver_.enable();
665
666 if (use_internal_timer) {
667 internal_timer_on_ = true;
668 ctrl_timer_.begin(
669 [this] {
670 control_step();
671 }, control_period_us_);
672 }
673 }

724 void update_sensors()
725 {
726 encoder_angle.update_angle(position_sensor_.read());
727 if (use_velocity_pll_) {
728 velocity_pll_.update(pos_sensor_dir_ * encoder_angle.get_full_angle());
729 shaft_angle_ = velocity_pll_.get_position();
730 shaft_velocity_ = velocity_pll_.get_velocity();
731 } else {
732 shaft_angle_ = pos_filter_.update(pos_sensor_dir_ * encoder_angle.get_full_angle());
733 shaft_velocity_ = vel_filter_cutoff_.update(vel_filter_.update(shaft_angle_));
734 }
735 electrical_angle_ = get_eangle(shaft_angle_);
736 phase_currents_ = cs_.get_phase_currents(true);
737 quaddirect_currents_ = phases_to_quaddirect<float>(phase_currents_, electrical_angle_);
738
739 // Thermal model update — I²R per-phase power from Park-frame currents.
740 // Factor of 1/2 converts amplitude-invariant peak current to per-phase RMS power:
741 // P_per_phase = (I_q² + I_d²) / 2 * R_phase
742 if (thermal_limiting_ && thermal_model_ != nullptr) {
743 float I_sq = quaddirect_currents_.q * quaddirect_currents_.q
744 + quaddirect_currents_.d * quaddirect_currents_.d;
745 thermal_model_->update(0.5f * I_sq * motor_.phase_R, control_period_s_);
746 }
747 }

760 void update_control()
761 {
762 switch (ctrl_mode_) {
763 case ControllerMode::DISABLE:
764 return;
765 case ControllerMode::OPEN_LOOP_VELOCITY:
766 open_loop_shaft_angle_ += (target_ * control_period_s_);
767 open_loop_shaft_velocity_ = target_;
768 {
769 // In this case, the target is a desired shaft velocity
770 // Everything is an estimate in open loop
771
772 float back_emf = motor_.kV * target_;
773
774 auto phase_volts = quaddirect_to_phases<float>({motor_.phase_R * motor_.SAFE_CURRENT + back_emf, 0.f},
775 get_eangle(open_loop_shaft_angle_));
776 auto cntr_volts = center_phase_voltages(phase_volts) + PhaseValues<float>{1.f, 1.f, 1.f};;
777 driver_.set_phase_voltages(cntr_volts);
778 }
779 return;
780
781 case ControllerMode::TORQUE:
782 {
783 // Add in cogging torque if needed
784 if (anticog_enable_) { target_ += torque_mapper_(shaft_angle_ - cogging_offset_); }
785
786 // Convert requested torque into a current request
787 float requested_current = target_ / motor_.kT;
788
789 // Hard clamp to motor peak current
790 requested_current = std::clamp(requested_current, -motor_.MAX_CURRENT, motor_.MAX_CURRENT);
791
792 // Thermal derating: soft clamp based on estimated winding temperature.
793 // Derating begins at 75% of thermal headroom above ambient and scales
794 // linearly to zero at T_max_winding.
795 if (thermal_limiting_ && thermal_model_ != nullptr) {
796 float I_limit = thermal_model_->get_current_limit(motor_.MAX_CURRENT);
797 requested_current = std::clamp(requested_current, -I_limit, I_limit);
798 }
799
800 // Generate desired current in QD frame, D = 0;
801 QuadDirectValues<float> desr_current{requested_current, 0.f};
802
803 PhaseValues<float> ctrl_volts{0.f, 0.f, 0.f};
804
805 // Pump controllers
806 if(feedforward_enable_) { ctrl_volts += feedforward(desr_current); }
807 if(feedback_enable_) { ctrl_volts += feedback(desr_current); }
808 if(back_emf_enable_) { ctrl_volts += back_emf_decoupler(); }
809
810 //Apply filter to these controllers
811 auto filtered_ctrl_volts = filter_phase_voltages(ctrl_volts);
812
813 // We don't want to filter this as this is a real thing
814 // Although, unless your motor had an insane pole pair count(> 500), the filters should not catch this
815 if(anticog_volt_enable_){
816 filtered_ctrl_volts += volt_mapper_(shaft_angle_ - cogging_offset_);
817 }
818
819 // Shift all voltages by 1 to avoid setting PWM pin to 0 as it will switch to digital and cause delays
820 const auto dr_volts = center_phase_voltages(filtered_ctrl_volts) + PhaseValues<float>{1.f, 1.f, 1.f};
821
822 last_phase_volts_ = dr_volts;
823
824 driver_.set_phase_voltages(dr_volts);
825
826 }
827 return;
828
829 default:
830 return;
831 }

1336 {
1337 // Convert desired dq current to abc phase frame, then apply RL inversion filter
1338 auto desr_phase_currents = quaddirect_to_phases<float>(desr_current, electrical_angle_);

1341
1356 PhaseValues<float> feedback(QuadDirectValues<float> desr_current)
1357 {
1358 QuadDirectValues<float> error = desr_current - quaddirect_currents_;
1359
1360 QuadDirectValues<float> pi_out = feedback_.update(error);
1361
1362 // Anti-windup: clamp PI output to the phase voltage ceiling and back-calculate
1363 // the integrator state so it doesn't accumulate beyond what the driver can apply.
1364 // This prevents integrator windup during current saturation (e.g. high speed,
1365 // mechanical lock, or thermal derating).
1366 QuadDirectValues<float> pi_clamped = {
1367 std::clamp(pi_out.q, -MAX_VOLT_, MAX_VOLT_),
1368 std::clamp(pi_out.d, -MAX_VOLT_, MAX_VOLT_)
1369 };
1370 if (pi_out.q != pi_clamped.q || pi_out.d != pi_clamped.d) {
1371 feedback_.set_output_state(pi_clamped);
1372 }
1373
1374 auto fb_phs_v = quaddirect_to_phases<float>(pi_clamped, electrical_angle_);
1375
1376 return filter_feedback_voltages(fb_phs_v);
1377 }

1439 PhaseValues<float> center_phase_voltages(PhaseValues<float> phase_volts) const
1440 {
1441 float _min = min(phase_volts.a, min(phase_volts.b, phase_volts.c));
1442
1443 PhaseValues<float> offset_volts{_min, _min, _min};
1444
1445 auto new_volts = phase_volts - offset_volts;
1446
1447 float _max = max(new_volts.a, max(new_volts.b, new_volts.c));
1448 float ratio = 1.f;
1449 if (_max > MAX_VOLT_) {
1450 ratio = MAX_VOLT_ / _max;
1451 }
1452 return new_volts * ratio;
1453 }
