128inline uint16_t make_read_cmd(uint16_t reg_addr)
129{
130 uint16_t word = static_cast<uint16_t>(RW_MASK | (reg_addr & DATA_MASK));
131 uint16_t p = static_cast<uint16_t>(__builtin_popcount(word) & 1u);
132 return static_cast<uint16_t>((p << 15u) | word);
133}
134
140inline bool parity_ok(uint16_t word)
141{
142 return (__builtin_popcount(word) & 1u) == 0u;
143}
144
151inline bool has_error_flag(uint16_t word)
152{
153 return (word & EF_MASK) != 0u;
154}
155
159inline uint16_t data_field(uint16_t word)
160{
161 return word & DATA_MASK;
162}

325 bool begin_dma(DMAChannel& ch)
326 {
327 lpspi_ = spi_enc_detail::lpspi_for(*spi_);
328 if (!lpspi_) return false;
329
330 dmamux_rx_ = spi_enc_detail::dmamux_rx_for(*spi_);
331 dma_ch_ = &ch;
332 dma_state_ = DMA_IDLE;
333
334 if (num_instances_ >= MAX_INSTANCES) return false;
335 instance_idx_ = static_cast<int8_t>(num_instances_);
336 instances_[num_instances_++] = this;
337
338 // Allocate the DMA channel once; trigger_async_read() will reconfigure it
339 // per frame without re-allocating.
340 ch.begin();
341 ch.disable();
342
343 return true;
344 }

346 // -----------------------------------------------------------------------
347 // DMA async read path
348 // -----------------------------------------------------------------------
349
368 void trigger_async_read()
369 {
370 if (!lpspi_ || !dma_ch_) return;
371 if (dma_state_ == DMA_FRAME1_ACTIVE ||
372 dma_state_ == DMA_FRAME2_ACTIVE) return;
373
374 dma_state_ = DMA_FRAME1_ACTIVE;
375
376 // Flush any stale data in the LPSPI RX FIFO before enabling DMA.
377 while (lpspi_->SR & LPSPI_SR_RDF) { (void)lpspi_->RDR; }
378
379 // Configure DMA for frame-1 RX: discard into dma_discard_.
380 _configure_dma_for_frame(dma_discard_, frame1_isrs_[instance_idx_]);
381
382 // Enable LPSPI RX DMA trigger.
383 lpspi_->DER |= LPSPI_DER_RDDE;
384
385 // Assert CS and start frame 1.
386 lpspi_->TCR = (lpspi_->TCR & 0xFFFFF000u) | LPSPI_TCR_FRAMESZ(15u);
387 digitalWriteFast(cs_, LOW);
388 lpspi_->TDR = read_cmd_;
389 }

432 float read() override
433 {
434 if (dma_ch_) {
435 // Spin-wait for any in-flight DMA to finish (avoids SPI bus conflict).
436 while (dma_state_ == DMA_FRAME1_ACTIVE || dma_state_ == DMA_FRAME2_ACTIVE) { /* ~µs */ }
437 if (dma_state_ == DMA_RESULT_READY) return read_async();
438 }
439 return _convert(read_raw());
440 }

456 uint16_t read_raw()
457 {
458 digitalWriteFast(cs_, LOW);
459 spi_->transfer16(read_cmd_);
460 digitalWriteFast(cs_, HIGH);
461 delayNanoseconds(400);
462 digitalWriteFast(cs_, LOW);
463 uint16_t raw = spi_->transfer16(0x0000u);
464 digitalWriteFast(cs_, HIGH);

485 float _convert(uint16_t raw) const
486 {
487 if (parity_check_enable_ && !ams5x47::parity_ok(raw)) return -1.f;
488 if (ef_check_enable_ && ams5x47::has_error_flag(raw)) return -1.f;
489
490 uint16_t counts = ams5x47::data_field(raw);
491 if (glitch_filter_enable_ &&
492 (counts == 0u || counts == static_cast<uint16_t>(max_count_ - 1)))
493 return -1.f;
494
495 return static_cast<float>(counts)
496 / static_cast<float>(max_count_ - 1)
497 * _2_PI_;
498 }

505 void _configure_dma_for_frame(volatile uint16_t& dest, void (*isr_fn)())
506 {
507 dma_ch_->disable();
508 #pragma GCC diagnostic push
509 #pragma GCC diagnostic ignored "-Wstrict-aliasing"
510 dma_ch_->source(*reinterpret_cast<volatile uint16_t*>(&lpspi_->RDR));
511 #pragma GCC diagnostic pop
512 dma_ch_->destination(dest);
513 dma_ch_->transferSize(2);
514 dma_ch_->transferCount(1);
515 dma_ch_->triggerAtHardwareEvent(dmamux_rx_);
516 dma_ch_->attachInterrupt(isr_fn);
517 dma_ch_->enable();
518 }

530 FASTRUN void handle_frame1_complete()
531 {
532 dma_ch_->clearInterrupt();
533 lpspi_->DER &= ~LPSPI_DER_RDDE;
534
535 digitalWriteFast(cs_, HIGH);
536 delayNanoseconds(400); // tCSn ≥ 400 ns (AS5047P datasheet §6.5)
537
538 // Reconfigure DMA for frame 2: capture angle into dma_result_.
539 dma_state_ = DMA_FRAME2_ACTIVE;
540 _configure_dma_for_frame(dma_result_, frame2_isrs_[instance_idx_]);
541 lpspi_->DER |= LPSPI_DER_RDDE;
542
543 // Assert CS and start frame 2.
544 lpspi_->TCR = (lpspi_->TCR & 0xFFFFF000u) | LPSPI_TCR_FRAMESZ(15u);
545 digitalWriteFast(cs_, LOW);
546 lpspi_->TDR = 0x0000u;
547 }
548
555 FASTRUN void handle_frame2_complete()
556 {
557 dma_ch_->clearInterrupt();
558 lpspi_->DER &= ~LPSPI_DER_RDDE;
559 digitalWriteFast(cs_, HIGH);
560 dma_state_ = DMA_RESULT_READY;
561 }
