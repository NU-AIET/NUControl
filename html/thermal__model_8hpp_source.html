197 void refresh()
198 {
199 float adc_raw = static_cast<float>(analogRead(pin_));
200
201 // Guard against divide-by-zero: clamp away from rail values
202 adc_raw = constrain(adc_raw, 1.f, adc_max_ - 1.f);
203
204 float V_adc = adc_raw / adc_max_ * V_supply_;
205 float R_ntc = R_series_ * V_adc / (V_supply_ - V_adc);
206
207 // B-value equation → temperature in Kelvin
208 float T_k = 1.f / (logf(R_ntc / R_nominal_) / beta_ + 1.f / T_nominal_k_);
209
210 cached_celsius_ = constrain(T_k - 273.15f, -40.f, 200.f);
211 }

477 void update(float P_winding_watts, float dt_s)
478 {
479 // Poll ambient sensor if attached. read_celsius() MUST be non-blocking
480 // (ISR-safe). For NTCThermistor, call refresh() from the main loop and
481 // pass the NTCThermistor pointer here — read_celsius() returns the cache.
482 if (ambient_sensor_ != nullptr) {
483 T_ambient_cached_ = ambient_sensor_->read_celsius();
484 }
485
486 // Use sensor readings as the current node temperatures when available;
487 // otherwise fall back to the ODE state. Loading volatile members into
488 // locals avoids repeated volatile reads in the arithmetic below.
489 const float T_w = winding_sensor_ ? winding_sensor_->read_celsius()
490 : static_cast<float>(T_winding_);
491 const float T_c = case_sensor_ ? case_sensor_->read_celsius()
492 : static_cast<float>(T_case_);
493
494 // Heat flows computed using best available node temperatures.
495 const float Q_out_w = (T_w - T_c) / R_th_wc_; // winding → case
496 const float Q_out_c = (T_c - T_ambient_cached_) / R_th_ca_; // case → ambient
497
498 // Sensor-attached nodes: store sensor reading (ground truth).
499 // Unsensed nodes: advance via forward-Euler ODE.
500 T_winding_ = winding_sensor_ ? T_w : T_w + (P_winding_watts - Q_out_w) / C_th_w_ * dt_s;
501 T_case_ = case_sensor_ ? T_c : T_c + (Q_out_w - Q_out_c) / C_th_c_ * dt_s;
502 }
503
516 float get_current_limit(float max_current) const
517 {
518 float T_warn = T_ambient_cached_ + 0.75f * (T_max_ - T_ambient_cached_);
519 if (T_winding_ <= T_warn) { return max_current; }
520 if (T_winding_ >= T_max_) { return 0.f; }
521 return max_current * (T_max_ - T_winding_) / (T_max_ - T_warn);
522 }
