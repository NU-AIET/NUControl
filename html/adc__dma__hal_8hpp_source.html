172 xbar_connect(XBARA1_IN_FLEXPWM2_PWM2_OUT_TRIG0, XBARA1_OUT_ADC_ETC_TRIG00);

287inline void configure_adc_etc_trig(uint8_t trig_idx, uint8_t adc_channel)
288{
289 // ADC_ETC is always clocked on iMXRT1062; no separate CCM gate needed.
290 ADC_ETC_CTRL &= ~ADC_ETC_CTRL_SOFTRST;
291 // Enable the bit corresponding to this trigger index
292 ADC_ETC_CTRL |= ADC_ETC_CTRL_TRIG_ENABLE(1u << trig_idx);
293
294 const uint32_t chain_entry =
295 ADC_ETC_TRIG_CHAIN_CSEL0(adc_channel) | // ADC channel selection
296 ADC_ETC_TRIG_CHAIN_HWTS0(1u) | // hardware trigger select
297 ADC_ETC_TRIG_CHAIN_IE0(0b01u) | // DMA request on conversion done
298 ADC_ETC_TRIG_CHAIN_B2B0; // back-to-back trigger enable
299
300 switch (trig_idx) {
301 case 0:
302 ADC_ETC_TRIG0_CTRL = ADC_ETC_TRIG_CTRL_TRIG_CHAIN(0u);
303 ADC_ETC_TRIG0_CHAIN_1_0 = chain_entry;
304 break;
305 case 1:
306 ADC_ETC_TRIG1_CTRL = ADC_ETC_TRIG_CTRL_TRIG_CHAIN(0u);
307 ADC_ETC_TRIG1_CHAIN_1_0 = chain_entry;
308 break;
309 case 2:
310 ADC_ETC_TRIG2_CTRL = ADC_ETC_TRIG_CTRL_TRIG_CHAIN(0u);
311 ADC_ETC_TRIG2_CHAIN_1_0 = chain_entry;
312 break;
313 default:
314 break;
315 }
316}

338inline void configure_adc_etc_trigger2(uint8_t adc_channel)
339{
340 // ADC_ETC is always clocked on iMXRT1062; no separate CCM gate needed.
341 ADC_ETC_CTRL &= ~ADC_ETC_CTRL_SOFTRST;
342 ADC_ETC_CTRL |= ADC_ETC_CTRL_TRIG_ENABLE(0b11u); // TRIG0 and TRIG1
343
344 ADC_ETC_TRIG1_CTRL = ADC_ETC_TRIG_CTRL_TRIG_CHAIN(0u);
345
346 ADC_ETC_TRIG1_CHAIN_1_0 =
347 ADC_ETC_TRIG_CHAIN_CSEL0(adc_channel) |
348 ADC_ETC_TRIG_CHAIN_HWTS0(1u) |
349 ADC_ETC_TRIG_CHAIN_IE0(0b01u) |
350 ADC_ETC_TRIG_CHAIN_B2B0;

377inline bool configure_adc1_hardware_trigger(uint8_t adc_channel)
378{
379 // Enable ADC1 clock (already on by default after analogRead(), but ensure it)
380 CCM_CCGR1 |= CCM_CCGR1_ADC1(CCM_CCGR_ON);
381
382 // 12-bit resolution, long sample time, no hardware averaging
383 ADC1_CFG = ADC_CFG_ADICLK(1) | // ADCK = IPG clock / 2
384 ADC_CFG_MODE(2) | // 12-bit (single-ended)
385 ADC_CFG_ADLSMP; // Long sample time (better noise rejection)
386
387 // Hardware trigger mode — ADTRG=1 in GC register
388 ADC1_CFG |= ADC_CFG_ADTRG; // Hardware trigger mode (bit 13 of CFG register)
389
390 // Select analog channel for hardware-triggered conversion
391 ADC1_HC0 = adc_channel;
392
393 return true;
394}

453inline bool configure_dma_for_adc1(DMAChannel & ch, volatile uint16_t & dest_buf)
454{
455 ch.begin();
456 // ADC1_R0 is a 32-bit register; the result occupies the low 16 bits at the same address.
457 // The cast is intentional hardware register access — suppress the strict-aliasing warning.
458 #pragma GCC diagnostic push
459 #pragma GCC diagnostic ignored "-Wstrict-aliasing"
460 ch.source(*reinterpret_cast<volatile uint16_t *>(&ADC1_R0)); // Source: ADC1 result register (low 16 bits)
461 #pragma GCC diagnostic pop
462 ch.destination(dest_buf); // Destination: caller's buffer
463 ch.transferSize(2); // 16-bit transfer
464 ch.transferCount(1); // 1 transfer per trigger
465
466 // Trigger on ADC_ETC conversion-done DMA request
467 ch.triggerAtHardwareEvent(DMAMUX_SOURCE_ADC_ETC);
468
469 ch.enable();
470 return true;
471}
