165inline void configure_center_aligned(FlexPWMPin pin)
166{
167 if (pin.module == nullptr) { return; }
168
169 auto & sm = pin.module->SM[pin.submodule];
170
171 // Step 1: halve VAL1 to preserve PWM frequency
172 // Edge-aligned period = VAL1 clocks; center-aligned period = 2×VAL1 clocks.
173 uint16_t half_val1 = sm.VAL1 / 2u;
174
175 // Step 2: scale duty compare values proportionally
176 // A channel uses VAL3; B channel uses VAL5. Both are scaled regardless of
177 // which channel this pin is, since both channels share the same submodule.
178 uint16_t val3_new = sm.VAL3 / 2u;
179 uint16_t val5_new = sm.VAL5 / 2u;
180
181 sm.VAL1 = half_val1;
182 sm.VAL3 = val3_new;
183 sm.VAL5 = val5_new;
184
185 // Step 3: set VAL0 to the counter peak value (= new VAL1).
186 // The output trigger (if enabled by configure_trigger_output) fires at
187 // the CNT == VAL0 compare, which is now the exact counter peak.
188 sm.VAL0 = half_val1;
189
190 // Step 4: enable center-aligned (up-down) counting.
191 // OR-in to preserve existing CTRL fields (prescaler, etc.) set by analogWriteFrequency().
192 sm.CTRL |= FLEXPWM_SMCTRL_FULL;
193
194 // Step 5: latch all shadow registers into active registers.
195 pin.module->MCTRL |= FLEXPWM_MCTRL_LDOK(1u << pin.submodule);
196}

218inline void configure_trigger_output(FlexPWMPin pin)
219{
220 if (pin.module == nullptr) { return; }
221
222 auto & sm = pin.module->SM[pin.submodule];
223
224 // OUT_TRIG_EN[0] = enable VAL0 compare output trigger
225 sm.TCTRL |= FLEXPWM_SMTCTRL_OUT_TRIG_EN(1u);
226
227 // Latch TCTRL change
228 pin.module->MCTRL |= FLEXPWM_MCTRL_LDOK(1u << pin.submodule);
229}

255inline void configure_submodule_sync(
256 IMXRT_FLEXPWM_t * module, uint8_t master_sm, uint8_t slave_sm)
257{
258 if (module == nullptr) { return; }
259
260 auto & sm = module->SM[slave_sm];
261
262 // CLK_SEL = 2 → use submodule 0's clock (or whichever master is SM[0]).
263 // If master_sm != 0, this approach has limitations; the RM §27.3.4 details
264 // that CLK_SEL=2 always uses SM[0], not an arbitrary master.
265 // For multi-motor setups where the master is not SM[0], use XBAR routing instead.
266 (void)master_sm; // Stored for documentation; CLK_SEL=2 always means SM[0]
267 sm.CTRL2 = (sm.CTRL2 & ~FLEXPWM_SMCTRL2_CLK_SEL(3u)) | FLEXPWM_SMCTRL2_CLK_SEL(2u);
268
269 // RELOAD_SEL: use master submodule's local SYNC signal to reload compare values
270 sm.CTRL2 |= FLEXPWM_SMCTRL2_RELOAD_SEL;
271
272 module->MCTRL |= FLEXPWM_MCTRL_LDOK(1u << slave_sm);
273}
