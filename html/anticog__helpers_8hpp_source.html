113 float get_cogging_torque(const float rads) const
114 {
115 if(steps_ == 0){
116 return 0.f;
117 }
118 float ang = normalize_angle(rads);
119 if(ang < 0) { ang += _2_PI_; }
120
121 const float idx = static_cast<float>(steps_) / _2_PI_ * ang;
122 const size_t idl = static_cast<size_t>(idx) % steps_;
123 const size_t idu = (idl + 1) % steps_;
124
125 const float alpha = ceilf(idx) - idx;
126 const float beta = 1.f - alpha;
127
128 return anticog_torque_map_.at(idl) * alpha + anticog_torque_map_.at(idu) * beta;
