122// ── Lock-free SPSC ring buffer ────────────────────────────────────────────────
123
125static constexpr size_t CAPACITY = 256;
126static_assert((CAPACITY & (CAPACITY - 1)) == 0, "CAPACITY must be a power of 2");
127
128struct RingBuffer {
129 Entry buf[CAPACITY];
130 std::atomic<size_t> head{0};
131 std::atomic<size_t> tail{0};
132 std::atomic<uint32_t> dropped{0};
133
136 bool push(Entry e) {
137 const size_t h = head.load(std::memory_order_relaxed);
138 const size_t next = (h + 1) & (CAPACITY - 1);
139 if (next == tail.load(std::memory_order_acquire)) {
140 dropped.fetch_add(1, std::memory_order_relaxed);
141 return false;
142 }
143 buf[h] = e;
144 head.store(next, std::memory_order_release);
145 return true;
146 }
147
150 bool pop(Entry& out) {
151 const size_t t = tail.load(std::memory_order_relaxed);
152 if (t == head.load(std::memory_order_acquire)) return false;
153 out = buf[t];
154 tail.store((t + 1) & (CAPACITY - 1), std::memory_order_release);
155 return true;
156 }
157};
