156 {
157 if (!inited_) { return false; }
158
159 pwm_hal::FlexPWMPin fpins[3] = {
160 pwm_hal::get_flexpwm_pin(static_cast<uint8_t>(pins_.a)),
161 pwm_hal::get_flexpwm_pin(static_cast<uint8_t>(pins_.b)),
162 pwm_hal::get_flexpwm_pin(static_cast<uint8_t>(pins_.c))
163 };
164
165 for (auto & fp : fpins) {
166 if (fp.module == nullptr) { return false; }
167 pwm_hal::configure_center_aligned(fp);
168 }
169
170 // Enable ADC trigger output on phase A submodule only.
171 // Phase A (pins_.a) is designated as the trigger source for the ADC chain.
172 pwm_hal::configure_trigger_output(fpins[0]);
173
174 // Configure intra-module SYNC if phase B or C share a module with phase A.
175 // Slaving them to phase A's submodule ensures they reach their counter peak
176 // at the same instant, maximising the quiet ADC sampling window.
177 if (fpins[1].module == fpins[0].module && fpins[1].submodule != fpins[0].submodule) {
178 pwm_hal::configure_submodule_sync(fpins[0].module, fpins[0].submodule, fpins[1].submodule);
179 }
180 if (fpins[2].module == fpins[0].module && fpins[2].submodule != fpins[0].submodule) {
181 pwm_hal::configure_submodule_sync(fpins[0].module, fpins[0].submodule, fpins[2].submodule);
182 }
183
184 // Store trigger source for caller retrieval
185 trigger_source_ = fpins[0];
186 center_aligned_ = true;
187 return true;
