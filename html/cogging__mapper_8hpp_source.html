301 void cogging_mapper()
302 {
303 controller_.update_sensors();
304 float error = normalize_angle(positions_.at(idx) - controller_.get_shaft_angle());
305 intl += (error * dt_);
306 intl = std::clamp(intl, -max_intl_, max_intl_);
307 float torque = kP_ * error - kD_ * controller_.get_shaft_velocity() + kI_ * intl;
308 torque = std::clamp(torque, -max_torque, max_torque);
309 controller_.set_target(torque);
310 controller_.update_control();
311 timeout_clk++;
312
313 // --- Timeout check ---
314 if (timeout_state_ && timeout_clk >= timeout_max_) {
315 torques_.at(idx) = timeout_val_;
316 phase_volts_.at(idx) = {timeout_val_, timeout_val_, timeout_val_};
317 reset_target_state();
318 Serial.println("Target Timedout");
319 idx++;
320 Serial.print("Target #");
321 Serial.println(idx);
322 }
323
324 // --- Lock detection ---
325 if (fabs(error) < pos_err_tol && !locked) {
326 locked = true;
327 settling_ = true;
328 settle_clk_ = 0;
329 }
330
331 if (locked) {
332 // Tolerance violated — reset to seeking
333 if (fabs(error) > pos_err_tol) {
334 locked = false;
335 settling_ = false;
336 settle_clk_ = 0;
337 torque_sum_ = 0.f;
338 volt_sum_ = {0.f, 0.f, 0.f};
339 clk_start = 0;
340 } else if (settling_) {
341 // Settle window: wait for ring-down before accumulating
342 settle_clk_++;
343 if (settle_clk_ >= settle_steps_) {
344 settling_ = false;
345 }
346 } else {
347 // Accumulation phase
348 torque_sum_ += torque * step_inv;
349 volt_sum_ += controller_.get_last_phasevolts() * step_inv;
350 clk_start++;
351 if (clk_start >= torque_steps_) {
352 torques_.at(idx) = torque_sum_;
353 phase_volts_.at(idx) = volt_sum_;
354 reset_target_state();
355 idx++;
356 Serial.print("Target #");
357 Serial.println(idx);
358 }
359 }
360 }
361
362 if (idx >= steps_) {
363 report_out();
364 }
365 }
