51 RingBuffer(size_t n, T fill = T{}) : buf_(n, fill), head_(0) {}
52
54 void push(const T& val)
55 {
56 if (buf_.empty()) return;
57 head_ = (head_ == 0 ? buf_.size() : head_) - 1;
58 buf_[head_] = val;
59 }
60
62 T& operator[](size_t age) { return buf_[(head_ + age) % buf_.size()]; }
63 const T& operator[](size_t age) const { return buf_[(head_ + age) % buf_.size()]; }
64
66 void fill_all(const T& val) { std::fill(buf_.begin(), buf_.end(), val); head_ = 0; }
67

73};

100template<typename T, typename G>
101class DiscreteFilter
102{
103public:
104 DiscreteFilter() = default;
105 ~DiscreteFilter() = default;
106
118 DiscreteFilter(std::vector<G> input_coeffs, std::vector<G> output_coeffs)
119 : input_coeffs_(input_coeffs),
120 output_coeffs_(output_coeffs),
121 inputs_num_(input_coeffs.size()),
122 outputs_num_(output_coeffs.size()),
123 inputs_(inputs_num_, static_cast<T>(0.f)),
124 outputs_(outputs_num_, static_cast<T>(0.f))
125 {}
126
137 virtual T update(T new_input)
138 {
139 inputs_.push(new_input);
140
141 T sum_ = static_cast<T>(0.f);
142 for (size_t i = 0; i < inputs_num_; ++i) {
143 sum_ += (input_coeffs_[i] * inputs_[i]);
144 }
145
146 if (outputs_num_ == 0) {
147 return sum_;
148 }
149
150 for (size_t j = 0; j < outputs_num_; ++j) {
151 sum_ -= (output_coeffs_[j] * outputs_[j]);
152 }
153
154 outputs_.push(sum_);
155
156 return sum_;
157 }
158
169 virtual void reset(T input_fill = static_cast<T>(0.f), T output_fill = static_cast<T>(0.f))
170 {
171 inputs_.fill_all(static_cast<T>(input_fill));
172 outputs_.fill_all(static_cast<T>(output_fill));
173 }
174
184 void set_output_state(T val)
185 {
186 if (!outputs_.empty()) outputs_[0] = val;
187 }
188
189private:
190 std::vector<G> input_coeffs_;
191 std::vector<G> output_coeffs_;
192
193 size_t inputs_num_;
194 size_t outputs_num_;
195 RingBuffer<T> inputs_;
196 RingBuffer<T> outputs_;
197};

224template<typename T>
225class Butterworth2nd : public DiscreteFilter<T, float>
226{
227public:
228 Butterworth2nd() = default;
229 ~Butterworth2nd() = default;
230
236 Butterworth2nd(float cutoff_hz, float sampling_hz)
237 : DiscreteFilter<T, float>(input_coeffs(cutoff_hz, sampling_hz),
238 output_coeffs(cutoff_hz, sampling_hz))
239 {}
240
241private:
243 std::vector<float> output_coeffs(float f_c, float f_s)
244 {
245 T alpha = 1.f / tanf(M_PI * f_c / f_s);
246 T beta = alpha * alpha + _SQRT_2_ * alpha + 1;
247
248 return {(-2.f * alpha * alpha + 2.f) / beta, (alpha * alpha - _SQRT_2_ * alpha + 1.f) / beta};
249
250 }
251
253 std::vector<float> input_coeffs(float f_c, float f_s)
254 {
255 T alpha = 1.f / tanf(M_PI * f_c / f_s);
256 T beta = alpha * alpha + _SQRT_2_ * alpha + 1;
257
258 return {1.f / beta, 2.f / beta, 1.f / beta};
259
260 }
